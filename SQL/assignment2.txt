Q4. Create the table Member again as per the schema description with the
    following constraints.
    a. Member_Id – Primary Key
    b. Membership_type - ‘Lifetime’,’ Annual’, ‘Half Yearly’,’ Quarterly’


mysql> create table member
    -> (member_id int primary key,
    -> member_name varchar(30),
    -> member_adress varchar(30),
    -> acc_open_date date,
    -> membership_type varchar(20) check (membership_type in('lifetime',
       'annual','half yearly', 'monthly','quarterly')),
    -> fees_paid int,
    -> max_books_allowed int,
